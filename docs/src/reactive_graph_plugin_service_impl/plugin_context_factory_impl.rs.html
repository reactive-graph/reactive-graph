105#[async_trait]
106#[component_alias]
107impl PluginContextFactory for PluginContextFactoryImpl {
108    fn construct_plugin_context(&self) {
109        // Type System
110        let component_manager = ComponentManagerDelegate::new(self.component_manager.clone());
111        let component_import_export_manager = ComponentImportExportManagerDelegate::new(self.component_import_export_manager.clone());
112        let entity_type_manager = EntityTypeManagerDelegate::new(self.entity_type_manager.clone());
113        let entity_type_import_export_manager = EntityTypeImportExportManagerDelegate::new(self.entity_type_import_export_manager.clone());
114        let relation_type_manager = RelationTypeManagerDelegate::new(self.relation_type_manager.clone());
115        let relation_type_import_export_manager = RelationTypeImportExportManagerDelegate::new(self.relation_type_import_export_manager.clone());
116        let flow_type_manager = FlowTypeManagerDelegate::new(self.flow_type_manager.clone());
117        let flow_type_import_export_manager = FlowTypeImportExportManagerDelegate::new(self.flow_type_import_export_manager.clone());
118        let type_system_provider_registry = TypeSystemProviderRegistryDelegate::new(&self.type_system_provider_registry);
119        // Instance System
120        let entity_instance_manager =
121            EntityInstanceManagerDelegate::new(self.component_manager.clone(), self.entity_type_manager.clone(), self.reactive_entity_manager.clone());
122        let relation_instance_manager =
123            RelationInstanceManagerDelegate::new(self.component_manager.clone(), self.relation_type_manager.clone(), self.reactive_relation_manager.clone());
124        let flow_instance_manager = FlowInstanceManagerDelegate::new(self.reactive_flow_manager.clone());
125        // Behaviour Registries
126        let entity_behaviour_registry = EntityBehaviourRegistryDelegate::new(
127            self.entity_behaviour_manager.clone(),
128            self.entity_behaviour_registry.clone(),
129            self.reactive_entity_manager.clone(),
130        );
131        let entity_component_behaviour_registry = EntityComponentBehaviourRegistryDelegate::new(
132            self.entity_component_behaviour_manager.clone(),
133            self.entity_component_behaviour_registry.clone(),
134            self.reactive_entity_manager.clone(),
135        );
136        let relation_behaviour_registry = RelationBehaviourRegistryDelegate::new(
137            self.relation_behaviour_manager.clone(),
138            self.relation_behaviour_registry.clone(),
139            self.reactive_relation_manager.clone(),
140        );
141        let relation_component_behaviour_registry = RelationComponentBehaviourRegistryDelegate::new(
142            self.relation_component_behaviour_manager.clone(),
143            self.relation_component_behaviour_registry.clone(),
144            self.reactive_relation_manager.clone(),
145        );
146        // GraphQL Services
147        let graphql_query_service = GraphQLQueryServiceDelegate::new(self.graphql_query_service.clone());
148        let web_resource_manager = WebResourceManagerDelegate::new(self.web_resource_manager.clone());
149        // System Services
150        let config_manager = ConfigManagerDelegate::new(self.config_manager.clone());
151        let type_system_event_manager = TypeSystemEventManagerDelegate::new(self.type_system_event_manager.clone());
152        let command_manager = CommandManagerDelegate::new(self.command_manager.clone());
153        let plugin_context = PluginContextImpl::new(
154            Arc::new(component_manager),
155            Arc::new(component_import_export_manager),
156            Arc::new(entity_type_manager),
157            Arc::new(entity_type_import_export_manager),
158            Arc::new(relation_type_manager),
159            Arc::new(relation_type_import_export_manager),
160            Arc::new(flow_type_manager),
161            Arc::new(flow_type_import_export_manager),
162            Arc::new(type_system_event_manager),
163            Arc::new(type_system_provider_registry),
164            Arc::new(entity_instance_manager),
165            Arc::new(relation_instance_manager),
166            Arc::new(flow_instance_manager),
167            Arc::new(entity_behaviour_registry),
168            Arc::new(entity_component_behaviour_registry),
169            Arc::new(relation_behaviour_registry),
170            Arc::new(relation_component_behaviour_registry),
171            Arc::new(config_manager),
172            Arc::new(graphql_query_service),
173            Arc::new(web_resource_manager),
174            Arc::new(command_manager),
175        );
176        let plugin_context = Arc::new(plugin_context);
177        let mut writer = self.plugin_context.write().unwrap();
178        let _ = writer.insert(plugin_context);
179    }
180
181    fn get(&self) -> Option<Arc<dyn PluginContext + Send + Sync>> {
182        let reader = self.plugin_context.read().unwrap();
183        if let Some(plugin_context) = reader.as_ref() {
184            return Some(plugin_context.clone());
185        }
186        None
187    }
188}

190#[async_trait]
191impl Lifecycle for PluginContextFactoryImpl {
192    async fn init(&self) {
193        self.construct_plugin_context();
194    }

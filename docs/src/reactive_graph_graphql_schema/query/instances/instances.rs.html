33    /// Search for entity instances.
34    ///
35    /// If an id is given, the entity instance with the given id will be returned.
36    ///
37    /// If an entity type is given, only entity instances of the given type are returned.
38    #[allow(clippy::too_many_arguments)]
39    async fn entities(
40        &self,
41        context: &Context<'_>,
42        #[graphql(desc = "Returns only the entity instance with the given id.")] id: Option<Uuid>,
43        #[graphql(desc = "Returns the entity instance with the given label.")] label: Option<String>,
44        #[graphql(
45            name = "type",
46            desc = "Filters the entity instances by the fully qualified namespace of the entity type.",
47            validator(custom = "NamespacedTypeValidator::new()")
48        )]
49        _type: Option<String>,
50        #[graphql(desc = "Filters the entity instances by applied components.")] components: Option<Vec<String>>,
51        #[graphql(desc = "Filters the entity instances by applied behaviours.")] behaviours: Option<Vec<String>>,
52        #[graphql(name = "properties", desc = "Query by properties.")] property_query: Option<Vec<GraphQLPropertyInstance>>,
53    ) -> Result<Vec<GraphQLEntityInstance>> {
54        let entity_instance_manager = context.data::<Arc<dyn ReactiveEntityManager + Send + Sync>>()?;
55        let entity_ty = EntityTypeId::parse_optional_namespace(_type)?;
56        let components = ComponentTypeIds::parse_optional_namespaces(components)?;
57        let behaviours = BehaviourTypeIds::parse_optional_namespaces(behaviours)?;
58        if let Some(id) = id {
59            let entity_instance = entity_instance_manager.get(id).map(|entity_instance| {
60                let entity_instance: GraphQLEntityInstance = entity_instance.into();
61                entity_instance
62            });
63            return if entity_instance.is_some() {
64                Ok(vec![entity_instance.unwrap()])
65            } else {
66                Ok(Vec::new())
67            };
68        }
69        if let Some(label) = label {
70            let entity_instance = entity_instance_manager.get_by_label(label.as_str()).map(|entity_instance| {
71                let entity_instance: GraphQLEntityInstance = entity_instance.into();
72                entity_instance
73            });
74            return if entity_instance.is_some() {
75                Ok(vec![entity_instance.unwrap()])
76            } else {
77                Ok(Vec::new())
78            };
79        }
80        let entities = entity_instance_manager
81            .get_all()
82            .iter()
83            .filter(|entity_instance| entity_ty.is_none() || entity_instance.ty == entity_ty.clone().unwrap())
84            .filter(|entity_instance| components.is_empty() || { entity_instance.is_all(&components) })
85            .filter(|entity_instance| behaviours.is_empty() || { entity_instance.behaves_as_all(&behaviours) })
86            .filter(|entity_instance| {
87                property_query.is_none() || {
88                    let property_query = property_query.clone().unwrap();
89                    if property_query.is_empty() {
90                        return true;
91                    }
92                    if entity_instance.properties.is_empty() {
93                        return false;
94                    }
95                    property_query
96                        .iter()
97                        .all(|property_query| match entity_instance.properties.get(property_query.name.as_str()) {
98                            Some(property_instance) => property_query.value == property_instance.get(),
99                            None => false,
100                        })
101                }
102            })
103            .map(|entity_instance| {
104                let entity_instance: GraphQLEntityInstance = entity_instance.clone().into();
105                entity_instance
106            })
107            .collect();
108        Ok(entities)
109    }

111    async fn count_entity_instances(
112        &self,
113        context: &Context<'_>,
114        #[graphql(
115            name = "type",
116            desc = "Counts the entity instances of the given entity type only.",
117            validator(custom = "NamespacedTypeValidator::new()")
118        )]
119        _type: Option<String>,
120        #[graphql(name = "component", desc = "Counts the entity instances which are composed by the given component only.")] component_type: Option<String>,
121        #[graphql(name = "behaviour", desc = "Counts the entity instances which behaves as the behaviour only.")] behaviour_type: Option<String>,
122    ) -> Result<usize> {
123        let entity_instance_manager = context.data::<Arc<dyn ReactiveEntityManager + Send + Sync>>()?;
124        let entity_ty = EntityTypeId::parse_optional_namespace(_type)?;
125        let component_ty: Option<ComponentTypeId> = ComponentTypeId::parse_optional_namespace(component_type)?;
126        let behaviour_ty: Option<BehaviourTypeId> = BehaviourTypeId::parse_optional_namespace(behaviour_type)?;
127        if entity_ty.is_none() && component_ty.is_none() && behaviour_ty.is_none() {
128            return Ok(entity_instance_manager.count());
129        }
130        if component_ty.is_none() && behaviour_ty.is_none() {
131            return Ok(entity_instance_manager.count_by_type(&entity_ty.unwrap()));
132        }
133        if entity_ty.is_none() && behaviour_ty.is_none() {
134            return Ok(entity_instance_manager.count_by_component(&component_ty.unwrap()));
135        }
136        if entity_ty.is_none() && component_ty.is_none() {
137            return Ok(entity_instance_manager.count_by_behaviour(&behaviour_ty.unwrap()));
138        }
139        let count = entity_instance_manager
140            .get_all()
141            .iter()
142            .filter(|e| entity_ty.is_none() || { e.ty == entity_ty.clone().unwrap() })
143            .filter(|e| component_ty.is_none() || e.is_a(&component_ty.clone().unwrap()))
144            .filter(|e| behaviour_ty.is_none() || e.behaves_as(&behaviour_ty.clone().unwrap()))
145            .count();
146        Ok(count)
147    }

149    /// Search for relations instances.
150    ///
151    /// Relation instances can be searched by relation type name, the entity type of the outbound
152    /// entity instance, the entity type of the inbound entity instance, the id of the outbound
153    /// entity instance or the id of the inbound entity instance. All of these filters can be
154    /// combined.
155    #[allow(clippy::too_many_arguments)]
156    async fn relations(
157        &self,
158        context: &Context<'_>,
159        #[graphql(desc = "Returns only the relation instance with the given id.")] id: Option<GraphQLRelationInstanceId>,
160        #[graphql(
161            name = "outboundEntity",
162            desc = "Filters the relation instances by the entity type of the outbound entity instance."
163        )]
164        outbound_entity_type: Option<String>,
165        #[graphql(
166            name = "outboundComponent",
167            desc = "Filters the relation instances by the component of the outbound entity instance."
168        )]
169        outbound_component_type: Option<String>,
170        #[graphql(desc = "Filters the relation instances by the id of the outbound entity instance")] outbound_id: Option<Uuid>,
171        #[graphql(
172            name = "type",
173            desc = "Filters the relation instances by relation type",
174            validator(custom = "NamespacedTypeValidator::new()")
175        )]
176        _type: Option<String>,
177        #[graphql(
178            name = "inboundEntity",
179            desc = "Filters the relation instances by the entity type of the inbound entity instance."
180        )]
181        inbound_entity_type: Option<String>,
182        #[graphql(
183            name = "inboundComponent",
184            desc = "Filters the relation instances by the component of the inbound entity instance."
185        )]
186        inbound_component_type: Option<String>,
187        #[graphql(desc = "Filters the relation instances by the id of the inbound entity instance")] inbound_id: Option<Uuid>,
188        #[graphql(desc = "Filters the relation instances by applied components.")] components: Option<Vec<String>>,
189        #[graphql(desc = "Filters the relation instances by applied behaviours.")] behaviours: Option<Vec<String>>,
190        #[graphql(name = "properties", desc = "Query by properties.")] property_query: Option<Vec<GraphQLPropertyInstance>>,
191    ) -> Result<Vec<GraphQLRelationInstance>> {
192        let relation_instance_manager = context.data::<Arc<dyn ReactiveRelationManager + Send + Sync>>()?;
193        if let Some(id) = id {
194            // let id = id.ty()?;
195            let id = id.try_into()?;
196            let relation_instance = relation_instance_manager.get(&id).map(|relation_instance| {
197                let relation_instance: GraphQLRelationInstance = relation_instance.into();
198                relation_instance
199            });
200            return if relation_instance.is_some() {
201                Ok(vec![relation_instance.unwrap()])
202            } else {
203                Ok(Vec::new())
204            };
205        }
206
207        let outbound_entity_ty = EntityTypeId::parse_optional_namespace(outbound_entity_type)?;
208        let outbound_component_ty = ComponentTypeId::parse_optional_namespace(outbound_component_type)?;
209        let relation_ty = RelationTypeId::parse_optional_namespace(_type)?;
210        let inbound_entity_ty = EntityTypeId::parse_optional_namespace(inbound_entity_type)?;
211        let inbound_component_ty = ComponentTypeId::parse_optional_namespace(inbound_component_type)?;
212        let components = ComponentTypeIds::parse_optional_namespaces(components)?;
213        // let components: Option<Vec<ComponentTypeId>> =
214        //     components.map(|components| components.iter().cloned().map(|component_ty| component_ty.into()).collect());
215        let behaviours = BehaviourTypeIds::parse_optional_namespaces(behaviours)?;
216        // let behaviours: Option<Vec<BehaviourTypeId>> =
217        //     behaviours.map(|behaviours| behaviours.iter().cloned().map(|behaviour_ty| behaviour_ty.into()).collect());
218        let relation_instances = relation_instance_manager
219            .get_all()
220            .iter()
221            .filter(|relation_instance| outbound_entity_ty.is_none() || relation_instance.outbound.ty == outbound_entity_ty.clone().unwrap())
222            .filter(|relation_instance| outbound_component_ty.is_none() || relation_instance.outbound.is_a(&outbound_component_ty.clone().unwrap()))
223            .filter(|relation_instance| relation_ty.is_none() || relation_instance.relation_type_id() == relation_ty.clone().unwrap())
224            .filter(|relation_instance| inbound_entity_ty.is_none() || relation_instance.inbound.ty == inbound_entity_ty.clone().unwrap())
225            .filter(|relation_instance| inbound_component_ty.is_none() || relation_instance.inbound.is_a(&inbound_component_ty.clone().unwrap()))
226            .filter(|relation_instance| outbound_id.is_none() || outbound_id.unwrap() == relation_instance.outbound.id)
227            .filter(|relation_instance| inbound_id.is_none() || inbound_id.unwrap() == relation_instance.inbound.id)
228            .filter(|relation_instance| {
229                components.is_empty() || {
230                    // let components = components.clone().unwrap();
231                    components.iter().all(|component_ty| relation_instance.is_a(&component_ty))
232                }
233            })
234            .filter(|relation_instance| {
235                behaviours.is_empty() || {
236                    // let behaviours = behaviours.clone().unwrap();
237                    behaviours.iter().all(|behaviour_ty| relation_instance.behaves_as(&behaviour_ty))
238                }
239            })
240            .filter(|relation_instance| {
241                property_query.is_none() || {
242                    let property_query = property_query.clone().unwrap();
243                    if property_query.is_empty() {
244                        return true;
245                    }
246                    if relation_instance.properties.is_empty() {
247                        return false;
248                    }
249                    property_query
250                        .iter()
251                        .all(|property_query| match relation_instance.properties.get(property_query.name.as_str()) {
252                            Some(property_instance) => property_query.value == property_instance.get(),
253                            None => false,
254                        })
255                }
256            })
257            .map(|relation_instance| {
258                let relation_instance: GraphQLRelationInstance = relation_instance.clone().into();
259                relation_instance
260            })
261            .collect();
262        Ok(relation_instances)
263    }

265    async fn count_relation_instances(
266        &self,
267        context: &Context<'_>,
268        #[graphql(
269            name = "type",
270            desc = "Counts the relation instances of the given relation type only.",
271            validator(custom = "NamespacedTypeValidator::new()")
272        )]
273        _type: Option<String>,
274        #[graphql(name = "component", desc = "Counts the relation instances which are composed by the given component only.")] component_type: Option<String>,
275        #[graphql(name = "behaviour", desc = "Counts the relation instances which behaves as the behaviour only.")] behaviour_type: Option<String>,
276    ) -> Result<usize> {
277        let relation_instance_manager = context.data::<Arc<dyn ReactiveRelationManager + Send + Sync>>()?;
278        let relation_ty = RelationTypeId::parse_optional_namespace(_type)?;
279        let component_ty: Option<ComponentTypeId> = ComponentTypeId::parse_optional_namespace(component_type)?;
280        let behaviour_ty: Option<BehaviourTypeId> = BehaviourTypeId::parse_optional_namespace(behaviour_type)?;
281        if relation_ty.is_none() && component_ty.is_none() && behaviour_ty.is_none() {
282            return Ok(relation_instance_manager.count());
283        }
284        if component_ty.is_none() && behaviour_ty.is_none() {
285            return Ok(relation_instance_manager.count_by_type(&relation_ty.unwrap()));
286        }
287        if relation_ty.is_none() && behaviour_ty.is_none() {
288            return Ok(relation_instance_manager.count_by_component(&component_ty.unwrap()));
289        }
290        if relation_ty.is_none() && component_ty.is_none() {
291            return Ok(relation_instance_manager.count_by_behaviour(&behaviour_ty.unwrap()));
292        }
293        let count = relation_instance_manager
294            .get_all()
295            .iter()
296            .filter(|reactive_relation| relation_ty.is_none() || { reactive_relation.relation_type_id() == relation_ty.clone().unwrap() })
297            .filter(|reactive_relation| component_ty.is_none() || reactive_relation.is_a(&component_ty.clone().unwrap()))
298            .filter(|reactive_relation| behaviour_ty.is_none() || reactive_relation.behaves_as(&behaviour_ty.clone().unwrap()))
299            .count();
300        Ok(count)
301    }

303    /// Search for flows and their contained instances.
304    async fn flows(
305        &self,
306        context: &Context<'_>,
307        #[graphql(desc = "Filters by the id of the flow")] id: Option<Uuid>,
308        #[graphql(desc = "Filters by the label of the flow")] label: Option<String>,
309        #[graphql(
310            name = "type",
311            desc = "Filters by the entity type of the flow instance",
312            validator(custom = "NamespacedTypeValidator::new()")
313        )]
314        _type: Option<String>,
315        // TODO: Add filter by contains entity instance
316        // TODO: Add filter by contains relation instance
317        // TODO: Add filter by property
318    ) -> Result<Vec<GraphQLFlowInstance>> {
319        let reactive_flow_manager = context.data::<Arc<dyn ReactiveFlowManager + Send + Sync>>()?;
320        if let Some(id) = id {
321            return match reactive_flow_manager.get(id).map(|flow| flow.into()) {
322                Some(flow) => Ok(vec![flow]),
323                None => Ok(Vec::new()),
324            };
325        }
326        if let Some(label) = label {
327            let flow = reactive_flow_manager.get_by_label(label.as_str()).map(|flow| {
328                let flow: GraphQLFlowInstance = flow.into();
329                flow
330            });
331            return if flow.is_some() { Ok(vec![flow.unwrap()]) } else { Ok(Vec::new()) };
332        }
333        let entity_ty = EntityTypeId::parse_optional_namespace(_type)?;
334        let flow_instances = reactive_flow_manager
335            .get_all()
336            .iter()
337            .filter(|flow_instance| match &entity_ty {
338                Some(entity_ty) => flow_instance.ty == entity_ty.clone().into(),
339                None => true,
340            })
341            .map(|flow| {
342                let flow: GraphQLFlowInstance = flow.clone().into();
343                flow
344            })
345            .collect();
346        Ok(flow_instances)
347    }

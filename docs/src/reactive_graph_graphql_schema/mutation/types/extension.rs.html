13#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, Eq, InputObject)]
14#[graphql(name = "ExtensionDefinition")]
15pub struct GraphQLExtensionDefinition {
16    /// The fully qualified namespace of the extension.
17    #[graphql(name = "type")]
18    pub _type: String,
19
20    /// Optionally, the fully qualified namespace of the entity type.
21    #[graphql(name = "entityType")]
22    pub entity_type: Option<String>,
23
24    /// The description of the extension.
25    pub description: String,
26
27    /// The extension as JSON representation.
28    pub extension: Value,
29}

31impl TryFrom<GraphQLExtensionDefinition> for Extension {
32    type Error = NamespacedTypeParseError;
33
34    fn try_from(extension: GraphQLExtensionDefinition) -> Result<Self, Self::Error> {
35        let extension_ty = ExtensionTypeId::from_str(&extension._type)?;
36        let entity_ty = EntityTypeId::parse_optional_namespace(extension.entity_type)?;
37        Ok(Extension {
38            ty: extension_ty,
39            entity_ty,
40            description: extension.description.clone(),
41            extension: extension.extension,
42        })
43    }
44}

46#[derive(Default)]
47pub struct GraphQLExtensionDefinitions(Extensions);

49impl GraphQLExtensionDefinitions {
50    pub fn new(extensions: Extensions) -> Self {
51        Self(extensions)
52    }
53
54    pub fn parse_definitions(extension_definitions: Vec<GraphQLExtensionDefinition>) -> Result<Extensions, NamespacedTypeParseError> {
55        GraphQLExtensionDefinitions::try_from(extension_definitions).map(|e| e.0)
56    }
57
58    pub fn parse_optional_definitions(extension_definitions: Option<Vec<GraphQLExtensionDefinition>>) -> Result<Extensions, NamespacedTypeParseError> {
59        match extension_definitions {
60            Some(extension_definitions) => GraphQLExtensionDefinitions::try_from(extension_definitions).map(|extension_definition| extension_definition.0),
61            None => Ok(Extensions::new()),
62        }
63    }

86impl TryFrom<Vec<GraphQLExtensionDefinition>> for GraphQLExtensionDefinitions {
87    type Error = NamespacedTypeParseError;
88
89    fn try_from(extension_definitions: Vec<GraphQLExtensionDefinition>) -> Result<Self, Self::Error> {
90        let extensions = Extensions::new();
91        for extension_definition in extension_definitions.into_iter() {
92            extensions.push(Extension::try_from(extension_definition)?);
93        }
94        Ok(GraphQLExtensionDefinitions::new(extensions))
95    }

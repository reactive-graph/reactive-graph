12#[derive(Serialize, Deserialize, Clone, Debug, OneofObject)]
13#[graphql(name = "InboundOutboundType")]
14pub enum GraphQLInboundOutboundType {
15    #[serde(rename = "component")]
16    Component(String),
17    #[serde(rename = "entity_type")]
18    EntityType(String),
19}
20
21impl TryFrom<GraphQLInboundOutboundType> for InboundOutboundType {
22    type Error = NamespacedTypeParseError;
23
24    fn try_from(ty: GraphQLInboundOutboundType) -> Result<Self, Self::Error> {
25        Ok(match ty {
26            GraphQLInboundOutboundType::Component(_type) => match _type.as_str() {
27                "*" => InboundOutboundType::Component(MatchingInboundOutboundType::Any),
28                _ => InboundOutboundType::Component(MatchingInboundOutboundType::NamespacedType(ComponentTypeId::from_str(&_type)?)),
29            },
30            GraphQLInboundOutboundType::EntityType(_type) => match _type.as_str() {
31                "*" => InboundOutboundType::EntityType(MatchingInboundOutboundType::Any),
32                _ => InboundOutboundType::EntityType(MatchingInboundOutboundType::NamespacedType(EntityTypeId::from_str(&_type)?)),
33            },
34        })
35    }
36}
37
38impl From<InboundOutboundType> for GraphQLInboundOutboundType {
39    fn from(ty: InboundOutboundType) -> Self {
40        match ty {
41            InboundOutboundType::Component(ty) => GraphQLInboundOutboundType::Component(ty.to_string()),
42            InboundOutboundType::EntityType(ty) => GraphQLInboundOutboundType::EntityType(ty.to_string()),
43        }
44    }
45}

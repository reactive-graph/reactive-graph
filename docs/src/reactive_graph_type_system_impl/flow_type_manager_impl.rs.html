73    fn register(&self, flow_type: FlowType) -> Result<FlowType, FlowTypeRegistrationError> {
74        let ty = flow_type.ty.clone();
75        self.namespaced_type_manager.register(ty.namespaced_type())?;
76        if self.has(&ty) {
77            return Err(FlowTypeRegistrationError::FlowTypeAlreadyExists(ty));
78        }
79        // Check that the entity types of every declared entity instance exists
80        for entity_ty in flow_type.uses_entity_types() {
81            if !self.entity_type_manager.has(&entity_ty) {
82                warn!("Flow type {ty} not fully initialized: No entity type {entity_ty}");
83            }
84        }
85        // Check that the relation type of every declared relation instance exists
86        for relation_ty in flow_type.uses_relation_types() {
87            if !self.relation_type_manager.has(&relation_ty) {
88                warn!("Flow type {ty} not fully initialized: No relation type named {relation_ty}");
89            }
90        }
91        // TODO: Check that entity instances referenced by a relation instance exists
92        // TODO: Check that relation instances outbound entity has correct entity_type
93        // TODO: Check that relation instances inbound entity has correct entity_type
94        self.flow_types.push(flow_type.clone());
95        debug!("Registered flow type {ty}");
96        self.event_manager.emit_event(TypeSystemEvent::FlowTypeCreated(ty));
97        Ok(flow_type)
98    }

140    fn create_flow_type(
141        &self,
142        ty: &FlowTypeId,
143        description: &str,
144        wrapper_entity_instance: EntityInstance,
145        entity_instances: EntityInstances,
146        relation_instances: RelationInstances,
147        variables: PropertyTypes,
148        extensions: Extensions,
149    ) -> Result<FlowType, FlowTypeCreationError> {
150        let flow_type = FlowType::builder()
151            .ty(ty)
152            .description(description)
153            .wrapper_entity_instance(wrapper_entity_instance)
154            .entity_instances(entity_instances)
155            .relation_instances(relation_instances)
156            .variables(variables)
157            .extensions(extensions)
158            .build();
159        self.register(flow_type).map_err(FlowTypeCreationError::RegistrationError)
160    }
161
162    fn update_description(&self, ty: &FlowTypeId, description: &str) -> Result<FlowType, FlowTypeUpdateError> {
163        if !self.has(ty) {
164            return Err(FlowTypeUpdateError::FlowTypeDoesNotExist(ty.clone()));
165        }
166        for mut flow_type in self.flow_types.iter_mut() {
167            if &flow_type.ty == ty {
168                flow_type.description = description.to_string();
169                // TODO: Notify about changed flow_type
170            }
171        }
172        self.get(ty).ok_or(FlowTypeUpdateError::FlowTypeDoesNotExist(ty.clone()))
173    }

251    fn delete(&self, ty: &FlowTypeId) -> Option<FlowType> {
252        self.namespaced_type_manager.delete(ty.as_ref());
253        self.flow_types.remove(ty).map(|(ty, flow_type)| {
254            self.event_manager.emit_event(TypeSystemEvent::FlowTypeDeleted(ty.clone()));
255            flow_type
256        })
257    }
258
259    fn validate(&self, ty: &FlowTypeId) -> bool {
260        if let Some(flow_type) = self.get(ty) {
261            return flow_type
262                .entity_instances
263                .iter()
264                .all(|entity_instance| self.entity_type_manager.validate(&entity_instance.ty))
265                && flow_type
266                    .relation_instances
267                    .iter()
268                    .all(|relation_instance| self.relation_type_manager.validate(&relation_instance.relation_type_id()));
269        }
270        false
271    }

26#[derive(Component)]
27pub struct FlowQueryObjectFactoryImpl {
28    entity_type_manager: Arc<dyn EntityTypeManager + Send + Sync>,
29    flow_type_manager: Arc<dyn FlowTypeManager + Send + Sync>,
30    json_schema_field_factory: Arc<dyn JsonSchemaFieldFactory + Send + Sync>,
31}
32
33impl FlowQueryObjectFactoryImpl {}
34
35#[async_trait]
36#[component_alias]
37impl FlowQueryObjectFactory for FlowQueryObjectFactoryImpl {
38    fn create_query_objects(&self) -> Vec<Object> {
39        let mut query_objects = Vec::<Object>::new();
40        for (_, flow_type) in self.flow_type_manager.get_all() {
41            query_objects.push(self.create_query_object(flow_type));
42        }
43        query_objects
44    }
45
46    fn create_query_object(&self, flow_type: FlowType) -> Object {
47        let object_type_name = object_type_name(&flow_type.ty, RootObjectType::Query);
48        trace!("Create query object {object_type_name} for {}", &flow_type.ty);
49        let mut object = Object::new(object_type_name)
50            .description(&flow_type.description)
51            .implement(INTERFACE_FLOW)
52            // Namespace path field
53            .field(namespace_path_field(flow_type.namespace()))
54            // ID field
55            .field(flow_id_field());
56        if let Some(field) = self
57            .json_schema_field_factory
58            .get_json_schema_field(FIELD_NAME_JSON_SCHEMA, &flow_type.type_definition())
59        {
60            object = object.field(field);
61        }
62
63        let entity_ty = flow_type.wrapper_type();
64        if let Some(entity_type) = self.entity_type_manager.get(&entity_ty) {
65            // `ComponentTypeIdContainer`s implements the interfaces of all components
66            // and add a component id field for each component
67            object = component_type_id_container_component_fields(&entity_type, object);
68            // PropertyTypeContainer adds property fields
69            object = property_container_property_fields::<EntityType, ReactiveFlow>(&entity_type, object);
70        }
71
72        object
73    }
74}

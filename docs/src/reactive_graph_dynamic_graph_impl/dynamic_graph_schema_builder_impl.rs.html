23pub struct DynamicGraphSchemaBuilderImpl {
24    interface_manager: Arc<dyn InterfaceManager + Send + Sync>,
25    query_object_manager: Arc<dyn QueryObjectManager + Send + Sync>,
26    query_namespace_object_tree_factory: Arc<dyn QueryNamespaceObjectTreeFactory + Send + Sync>,
27    mutation_object_manager: Arc<dyn MutationObjectManager + Send + Sync>,
28    mutation_namespace_object_tree_factory: Arc<dyn MutationNamespaceObjectTreeFactory + Send + Sync>,

30    schema_builder_manager: Arc<dyn SchemaBuilderManager + Send + Sync>,
31    union_manager: Arc<dyn UnionManager + Send + Sync>,
32}
33
34impl DynamicGraphSchemaBuilderImpl {}
35
36#[async_trait]
37#[component_alias]
38impl DynamicGraphSchemaBuilder for DynamicGraphSchemaBuilderImpl {
39    fn build_dynamic_schema(&self) -> Result<Schema, SchemaError> {
40        let mut schema = self.schema_builder_manager.get_schema_builder();
41        schema = register_all(schema, self.scalar_manager.get_scalars());
42        schema = register_all(schema, self.interface_manager.get_interfaces());
43        // query
44        schema = register_all(schema, self.query_object_manager.get_query_objects());
45        schema = register_all(schema, self.query_namespace_object_tree_factory.get_namespace_objects());
46        // schema = schema.register(self.query_namespace_object_tree_factory.create_root_object());
47        // mutation
48        schema = register_all(schema, self.mutation_object_manager.get_mutation_objects());
49        schema = register_all(schema, self.mutation_namespace_object_tree_factory.get_namespace_objects());
50        // schema = schema.register(self.mutation_namespace_object_tree_factory.create_root_object());
51        schema = register_all(schema, self.union_manager.get_unions());
52        schema.finish()
53    }
54}
55
56#[async_trait]
57impl Lifecycle for DynamicGraphSchemaBuilderImpl {}

59fn register_all(mut schema: SchemaBuilder, tys: Vec<impl Into<Type> + Debug>) -> SchemaBuilder {
60    for ty in tys {
61        let ty = ty.into();
62        trace!("Registering dynamic graph type {:?}", ty);
63        schema = schema.register(ty);
64    }
65    schema

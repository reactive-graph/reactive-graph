35#[derive(Component)]
36pub struct EntityMutationFieldFactoryImpl {
37    entity_type_manager: Arc<dyn EntityTypeManager + Send + Sync>,
38    entity_query_field_factory: Arc<dyn EntityQueryFieldFactory + Send + Sync>,
39}
40
41#[async_trait]
42#[component_alias]
43impl EntityMutationFieldFactory for EntityMutationFieldFactoryImpl {
44    fn create_mutation_fields(&self, namespace: &Namespace) -> Vec<Field> {
45        let mut fields = Vec::new();
46        for entity_type in self.entity_type_manager.get_by_namespace(&namespace).iter().sorted_by(sort_by_key) {
47            fields.push(self.entity_query_field_factory.create_query_field(&entity_type, Mutation));
48            fields.push(self.create_creation_field(&entity_type));
49        }
50        fields
51    }
52
53    fn create_creation_field(&self, entity_type: &EntityType) -> Field {
54        let ty = entity_type.ty.clone();
55        let entity_type_inner = entity_type.clone();
56        let field_name = format!("{}{}", FIELD_NAME_PREFIX_CREATE, ty.type_name().to_case(Pascal));
57        let mut field = Field::new(field_name, object_type_ref(&entity_type.ty, Query), move |ctx| {
58            let ty = ty.clone();
59            let entity_type = entity_type_inner.clone();
60            FieldFuture::new(async move {
61                let reactive_entity_manager = ctx.data::<Arc<dyn ReactiveEntityManager + Send + Sync>>()?;
62                let id = if let Some(id) = ctx.args.get(QUERY_ARGUMENT_ID) {
63                    let id = Uuid::from_str(id.string()?)?;
64                    if reactive_entity_manager.has(id) {
65                        return Err(CreateEntityInstanceError::EntityInstanceAlreadyExist(id).into());
66                    }
67                    id
68                } else {
69                    Uuid::new_v4()
70                };
71                let properties = create_properties_from_field_arguments(&ctx, &entity_type.properties, false)?;
72                let properties = ReactiveProperties::new_with_id_from_properties(id, properties);
73                let reactive_entity = ReactiveEntity::builder().ty(&ty).id(id).properties(properties).build();
74                if let Ok(reactive_entity) = reactive_entity_manager.register_reactive_instance(reactive_entity) {
75                    return Ok(Some(FieldValue::owned_any(reactive_entity)));
76                }
77                Ok(None)
78            })
79        })
80        .description(format!("Create a new {} entity", object_type_name(&entity_type.ty, Query)))
81        .argument(query_argument_id());
82        field = add_property_type_container_properties_as_field_arguments(field, entity_type, false, false);
83        field
84    }
85}

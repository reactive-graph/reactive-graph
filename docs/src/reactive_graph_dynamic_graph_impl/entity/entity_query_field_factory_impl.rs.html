38pub struct EntityQueryFieldFactoryImpl {
39    entity_type_manager: Arc<dyn EntityTypeManager + Send + Sync>,
40    json_schema_field_factory: Arc<dyn JsonSchemaFieldFactory + Send + Sync>,

43#[async_trait]
44#[component_alias]
45impl EntityQueryFieldFactory for EntityQueryFieldFactoryImpl {
46    fn create_query_fields(&self, namespace: &Namespace) -> Vec<Field> {
47        let mut fields = Vec::new();
48        for entity_type in self.entity_type_manager.get_by_namespace(&namespace).iter().sorted_by(sort_by_key) {
49            fields.push(self.create_query_field(&entity_type, RootObjectType::Query));
50            if let Some(field) = self
51                .json_schema_field_factory
52                .get_json_schema_field(&json_schema_field_name(entity_type.key()), &entity_type.type_definition())
53            {
54                fields.push(field);
55            }
56        }
57        fields
58    }
59
60    fn create_query_field(&self, entity_type: &EntityType, root_object_type: RootObjectType) -> Field {
61        let ty = entity_type.ty.clone();
62        let entity_type_inner = entity_type.clone();
63        // Within the namespace object the short type name is used
64        let field_name = ty.type_name().to_case(Pascal);
65        let mut field = Field::new(field_name, object_type_ref_list(&ty, root_object_type), move |ctx| {
66            let ty = ty.clone();
67            let entity_type = entity_type_inner.clone();
68            FieldFuture::new(async move {
69                let reactive_entity_manager = ctx.data::<Arc<dyn ReactiveEntityManager + Send + Sync>>()?;
70
71                // Select single instance by id and type
72                if let Ok(id) = ctx.args.try_get(QUERY_ARGUMENT_ID) {
73                    let id = Uuid::from_str(id.string()?)?;
74                    let reactive_entity = reactive_entity_manager
75                        .get(id)
76                        .ok_or(QueryEntityInstanceError::EntityInstanceDoesNotExist(id))
77                        .map_err(Error::new_with_source)?;
78                    if reactive_entity.ty != ty {
79                        return Err(QueryEntityInstanceError::EntityInstanceIsNotOfType(id, ty.clone()).into());
80                    }
81                    return Ok(Some(FieldValue::list(vec![FieldValue::owned_any(reactive_entity)])));
82                }
83                // Select single instance by label and type
84                if let Ok(label) = ctx.args.try_get(QUERY_ARGUMENT_LABEL) {
85                    let label = label.string()?.to_string();
86                    let reactive_entity = reactive_entity_manager
87                        .get_by_label(&label)
88                        .ok_or(QueryEntityInstanceError::EntityInstanceWithLabelDoesNotExist(label))?;
89                    if reactive_entity.ty != ty {
90                        return Err(QueryEntityInstanceError::EntityInstanceIsNotOfType(reactive_entity.id, ty.clone()).into());
91                    }
92                    return Ok(Some(FieldValue::list(vec![FieldValue::owned_any(reactive_entity)])));
93                }
94                // Select instances by ids and type and properties
95                if let Ok(ids) = ctx.args.try_get(QUERY_ARGUMENT_IDS) {
96                    let mut instances = Vec::new();
97                    for id in ids
98                        .list()?
99                        .iter()
100                        .filter_map(|id| id.string().map(str::to_string).ok())
101                        .filter_map(|id| Uuid::from_str(&id).ok())
102                    {
103                        if let Some(reactive_entity) = reactive_entity_manager.get(id) {
104                            if reactive_entity.ty != ty {
105                                return Err(QueryEntityInstanceError::EntityInstanceIsNotOfType(reactive_entity.id, ty.clone()).into());
106                            }
107                            instances.push(reactive_entity);
108                        }
109                    }
110                    let instances = filter_instances_by_properties(&ctx, &entity_type, instances);
111                    return Ok(Some(FieldValue::list(instances.into_iter().map(FieldValue::owned_any))));
112                }
113                // Select instances by type only
114                let instances = reactive_entity_manager.get_by_type(&entity_type.ty);
115                let instances = filter_instances_by_properties(&ctx, &entity_type, instances);
116                Ok(Some(FieldValue::list(instances.into_iter().map(FieldValue::owned_any))))
117            })
118        })
119        .description(entity_type.description.clone())
120        .argument(query_argument_id())
121        .argument(query_argument_ids())
122        .argument(query_argument_label());
123        field = add_property_type_container_properties_as_field_arguments(field, entity_type, true, true);
124        field
125    }
126}

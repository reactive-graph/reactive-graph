37#[derive(Component)]
38pub struct RelationMutationObjectFactoryImpl {
39    relation_type_manager: Arc<dyn RelationTypeManager + Send + Sync>,
40    json_schema_field_factory: Arc<dyn JsonSchemaFieldFactory + Send + Sync>,
41}
42
43#[async_trait]
44#[component_alias]
45impl RelationMutationObjectFactory for RelationMutationObjectFactoryImpl {
46    fn create_mutation_objects(&self) -> Vec<Object> {
47        let mut objects = Vec::new();
48        for (_, relation_type) in self.relation_type_manager.get_all() {
49            objects.push(self.create_mutation_object(relation_type));
50        }
51        objects
52    }
53
54    fn create_mutation_object(&self, relation_type: RelationType) -> Object {
55        let object_type_name = object_type_name(&relation_type.ty, RootObjectType::Mutation);
56        trace!("Create mutation object for entity type {object_type_name}");
57        let mut object = Object::new(object_type_name)
58            .description(&relation_type.description)
59            // Namespace path field
60            .field(namespace_path_field(relation_type.namespace()))
61            // Relation ID field
62            .field(relation_id_field())
63            // Instance ID field
64            .field(relation_instance_id_field());
65        if let Some(field) = self
66            .json_schema_field_factory
67            .get_json_schema_field(FIELD_NAME_JSON_SCHEMA, &relation_type.type_definition())
68        {
69            object = object.field(field);
70        }
71        if let Some(update_field) = self.create_update_field(&relation_type) {
72            object = object.field(update_field);
73        }
74        if let Some(trigger_field) = self.create_trigger_field(&relation_type) {
75            object = object.field(trigger_field);
76        }
77        object = object.field(self.create_export_field());
78        object = object.field(self.create_delete_field());
79        object
80    }
81
82    fn create_update_field(&self, relation_type: &RelationType) -> Option<Field> {
83        let relation_type_inner = relation_type.clone();
84        let update_field = Field::new(FIELD_NAME_UPDATE, object_type_ref(&relation_type.ty, RootObjectType::Query), move |ctx| {
85            let relation_type = relation_type_inner.clone();
86            FieldFuture::new(async move {
87                let reactive_relation = ctx.parent_value.try_downcast_ref::<ReactiveRelation>()?;
88                // First validate all input fields for mutability and correct datatype
89                property_container_validate_input_fields(&relation_type, &ctx)?;
90                // Set properties
91                property_container_update_properties(&relation_type, reactive_relation, &ctx)?;
92                Ok(Some(FieldValue::owned_any(reactive_relation.clone())))
93            })
94        })
95        .description("Updates the entity instance");
96        property_container_update_field_arguments(relation_type, update_field)
97    }
98
99    fn create_trigger_field(&self, relation_type: &RelationType) -> Option<Field> {
100        create_trigger_field::<ReactiveRelation, RelationType>(&relation_type)
101    }
102
103    fn create_export_field(&self) -> Field {
104        create_export_field::<ReactiveRelation>()
105    }
106
107    fn create_delete_field(&self) -> Field {
108        Field::new(FIELD_NAME_DELETE, TYPE_REF_ID.clone(), move |ctx| {
109            FieldFuture::new(async move {
110                let relation_instance_manager = ctx.data::<Arc<dyn ReactiveRelationManager + Send + Sync>>()?;
111                let reactive_relation = ctx.parent_value.try_downcast_ref::<ReactiveRelation>()?;
112                let id = reactive_relation.id();
113                Ok(if relation_instance_manager.delete(&id) {
114                    Some(FieldValue::value(ID(id.to_string())))
115                } else {
116                    None
117                })
118            })
119        })
120    }
121}

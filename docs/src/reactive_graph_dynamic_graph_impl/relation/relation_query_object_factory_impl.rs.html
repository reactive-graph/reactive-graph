43impl RelationQueryObjectFactoryImpl {
44    fn inbound_outbound_fields(&self, relation_type: &RelationType, mut object: Object) -> Object {
45        // Outbound fields
46        for field in self.inbound_outbound_field(&relation_type, InboundOutboundDirection::Outbound, &relation_type.outbound_type) {
47            object = object.field(field);
48        }
49        // Inbound fields
50        for field in self.inbound_outbound_field(&relation_type, InboundOutboundDirection::Inbound, &relation_type.inbound_type) {
51            object = object.field(field);
52        }
53        object
54    }
55
56    fn inbound_outbound_field(&self, relation_type: &RelationType, dir: InboundOutboundDirection, ty: &InboundOutboundType) -> Vec<Field> {
57        // Look up field names and descriptions in extensions
58        let field_names = get_dynamic_graph_field_names(relation_type);
59        let field_descriptions = get_dynamic_graph_field_descriptions(relation_type);
60        let (field_name, field_description) = match dir {
61            InboundOutboundDirection::Outbound => (field_names.from_relation_to_outbound_entity, field_descriptions.from_relation_to_outbound_entity),
62            InboundOutboundDirection::Inbound => (field_names.from_relation_to_inbound_entity, field_descriptions.from_relation_to_inbound_entity),
63        };
64
65        match ty {
66            InboundOutboundType::EntityType(ty) => match ty {
67                MatchingInboundOutboundType::NamespacedType(ty) => {
68                    // If no field name was given via extensions, the field name is generated using
69                    // the direction and the fully qualified type name. The direction is necessary
70                    // because the outbound and inbound fields could be of the same type and every
71                    // field name must be unique.
72                    let field_name = field_name.unwrap_or_else(|| format!("{dir}_{}", ty.fully_qualified_type_name()));
73                    let type_name = object_type_name(ty, RootObjectType::Query);
74                    vec![create_inbound_outbound_field(dir, &type_name, &field_name, field_description)]
75                }
76                MatchingInboundOutboundType::Any => {
77                    // Returns the union which represents all entities. If no field name was given
78                    // via extensions, the field name is the direction (inbound or outbound).
79                    let field_name = field_name.unwrap_or(format!("{dir}"));
80                    vec![create_inbound_outbound_field(dir, UNION_ALL_ENTITIES, &field_name, field_description)]
81                }
82            },
83            InboundOutboundType::Component(ty) => match ty {
84                MatchingInboundOutboundType::NamespacedType(ty) => {
85                    // If no field name was given via extensions, the field name is generated using
86                    // the direction and the fully qualified type name. The direction is necessary
87                    // because the outbound and inbound fields could be of the same type and every
88                    // field name must be unique.
89                    let field_name = field_name.unwrap_or_else(|| format!("{dir}_{}", ty.fully_qualified_type_name()));
90                    let type_name = object_type_name(ty, RootObjectType::Interface);
91                    vec![create_inbound_outbound_field(dir, &type_name, &field_name, field_description)]
92                }
93                MatchingInboundOutboundType::Any => {
94                    self.component_manager
95                        .get_type_ids()
96                        .into_iter()
97                        .map(|ty| {
98                            // As we return multiple fields and every field name must be unique,
99                            // we have to prefix the direction AND use the type name
100                            let field_name = format!("{dir}_{}", ty.fully_qualified_type_name());
101                            let type_name = object_type_name(&ty, RootObjectType::Interface);
102                            create_inbound_outbound_field(dir.clone(), &type_name, &field_name, None)
103                        })
104                        .collect()
105                }
106            },
107        }
108    }
109}

112#[component_alias]
113impl RelationQueryObjectFactory for RelationQueryObjectFactoryImpl {
114    fn create_query_objects(&self) -> Vec<Object> {
115        let mut query_objects = Vec::<Object>::new();
116        for (_, relation_type) in self.relation_type_manager.get_all() {
117            query_objects.push(self.create_query_object(relation_type));
118        }
119        query_objects
120    }
121
122    fn create_query_object(&self, relation_type: RelationType) -> Object {
123        let object_type_name = object_type_name(&relation_type.ty, RootObjectType::Query);
124        trace!("Create query object {object_type_name} for {}", &relation_type.ty);
125        let mut object = Object::new(object_type_name)
126            .description(&relation_type.description)
127            .implement(INTERFACE_RELATION)
128            // Namespace path field
129            .field(namespace_path_field(relation_type.namespace()))
130            // Relation ID field
131            .field(relation_id_field())
132            // Instance ID field
133            .field(relation_instance_id_field());
134        if let Some(field) = self
135            .json_schema_field_factory
136            .get_json_schema_field(FIELD_NAME_JSON_SCHEMA, &relation_type.type_definition())
137        {
138            object = object.field(field);
139        }
140        // `ComponentTypeIdContainer`s implements the interfaces of all components
141        // and add a component id field for each component
142        object = component_type_id_container_component_fields(&relation_type, object);
143        // PropertyTypeContainer adds property fields
144        object = property_container_property_fields::<RelationType, ReactiveRelation>(&relation_type, object);
145        // Inbound and outbound Fields
146        object = self.inbound_outbound_fields(&relation_type, object);
147        object
148    }
149}

154pub fn create_inbound_outbound_field(dir: InboundOutboundDirection, type_name: &str, field_name: &str, field_description: Option<String>) -> Field {
155    let mut field = Field::new(field_name, TypeRef::named_nn(type_name), move |ctx| {
156        FieldFuture::new({
157            let inner_dir = dir.clone();
158            async move {
159                let dir = inner_dir.clone();
160                let reactive_relation = ctx.parent_value.try_downcast_ref::<ReactiveRelation>()?;
161                let reactive_entity = match dir {
162                    InboundOutboundDirection::Outbound => reactive_relation.outbound.clone(),
163                    InboundOutboundDirection::Inbound => reactive_relation.inbound.clone(),
164                };
165                Ok(Some(FieldValue::owned_any(reactive_entity)))
166            }
167        })
168    });
169    if let Some(field_description) = field_description {
170        field = field.description(field_description);
171    }
172    field
173}

35pub struct RelationQueryFieldFactoryImpl {
36    relation_type_manager: Arc<dyn RelationTypeManager + Send + Sync>,
37    json_schema_field_factory: Arc<dyn JsonSchemaFieldFactory + Send + Sync>,

40#[async_trait]
41#[component_alias]
42impl RelationQueryFieldFactory for RelationQueryFieldFactoryImpl {
43    fn create_query_fields(&self, namespace: &Namespace) -> Vec<Field> {
44        let mut fields = Vec::new();
45        for relation_type in self.relation_type_manager.get_by_namespace(&namespace).iter().sorted_by(sort_by_key) {
46            fields.push(self.create_query_field(&relation_type, RootObjectType::Query));
47            if let Some(field) = self
48                .json_schema_field_factory
49                .get_json_schema_field(&json_schema_field_name(relation_type.key()), &relation_type.type_definition())
50            {
51                fields.push(field);
52            }
53        }
54        fields
55    }
56
57    fn create_query_field(&self, relation_type: &RelationType, root_object_type: RootObjectType) -> Field {
58        let ty = relation_type.ty.clone();
59        let relation_type_inner = relation_type.clone();
60        let field_name = ty.type_name().to_case(Pascal);
61        let mut field = Field::new(field_name, object_type_ref_list(&ty, root_object_type), move |ctx| {
62            let relation_type = relation_type_inner.clone();
63            FieldFuture::new(async move {
64                let reactive_relation_manager = ctx.data::<Arc<dyn ReactiveRelationManager + Send + Sync>>()?;
65                let reactive_relations = reactive_relation_manager.get_by_type(&relation_type.ty);
66                let reactive_relations = filter_instances_by_properties(&ctx, &relation_type, reactive_relations);
67                let reactive_relations = filter_instances_by_outbound_id(&ctx, reactive_relations)?;
68                let reactive_relations = filter_instances_by_inbound_id(&ctx, reactive_relations)?;
69                Ok(Some(FieldValue::list(reactive_relations.into_iter().map(FieldValue::owned_any))))
70            })
71        })
72        .description(relation_type.description.clone())
73        .argument(query_argument_outbound_id())
74        .argument(query_argument_inbound_id());
75        field = add_property_type_container_properties_as_field_arguments(field, relation_type, true, true);
76        field
77    }
78}

80#[async_trait]
81impl Lifecycle for RelationQueryFieldFactoryImpl {}

83fn filter_instances_by_outbound_id(ctx: &ResolverContext, mut instances: Vec<ReactiveRelation>) -> Result<Vec<ReactiveRelation>, Error> {
84    if let Ok(outbound_id) = ctx.args.try_get("outbound_id") {
85        let outbound_id = Uuid::from_str(outbound_id.string()?)?;
86        instances.retain(|reactive_relation| reactive_relation.outbound.id == outbound_id);
87    }
88    Ok(instances)
89}
90
91fn filter_instances_by_inbound_id(ctx: &ResolverContext, mut instances: Vec<ReactiveRelation>) -> Result<Vec<ReactiveRelation>, Error> {
92    if let Ok(inbound_id) = ctx.args.try_get("inbound_id") {
93        let inbound_id = Uuid::from_str(inbound_id.string()?)?;
94        instances.retain(|reactive_relation| reactive_relation.outbound.id == inbound_id);
95    }
96    Ok(instances)
97}

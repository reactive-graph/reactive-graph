12/// Returns a list of property instances from the field arguments.
13///
14/// Returns an error if a property doesn't exist in the entity type or relation type.
15///
16/// First, initializes the properties with the default values of the property types of
17/// an entity type or relation type. Next, overwrites the properties with the value
18/// provided by the field arguments.
19pub fn create_properties_from_field_arguments(ctx: &ResolverContext, property_types: &PropertyTypes, optional: bool) -> Result<PropertyInstances, Error> {
20    let properties = PropertyInstances::new_from_property_types_with_defaults(property_types);
21    for property in property_types.iter() {
22        let Ok(field_arg_value) = ctx.args.try_get(property.key()) else {
23            if optional {
24                continue;
25            } else {
26                return Err(PropertyArgumentMissingError::PropertyArgumentMissing(property.key().clone()).into());
27            }
28        };
29        match &property.data_type {
30            DataType::Null => {
31                return Err(PropertyDataTypeError::NullIsNotAValidDataType(property.key().clone()).into());
32            }
33            DataType::Bool => {
34                properties.insert(property.key().clone(), Value::Bool(field_arg_value.boolean()?));
35            }
36            DataType::Number => {
37                if let Ok(value) = field_arg_value.i64() {
38                    properties.insert(property.key().clone(), json!(value));
39                } else if let Ok(value) = field_arg_value.u64() {
40                    properties.insert(property.key().clone(), json!(value));
41                } else if let Ok(value) = field_arg_value.f64() {
42                    properties.insert(property.key().clone(), json!(value));
43                } else {
44                    return Err(PropertyDataTypeError::ValueIsNotOfTheExpectedDataType(property.name.clone(), property.data_type, DataType::Number).into());
45                }
46            }
47            DataType::String => {
48                properties.insert(property.key().clone(), Value::String(field_arg_value.string()?.to_string()));
49            }
50            DataType::Array => {
51                let _ = field_arg_value.list()?;
52                let value = field_arg_value.deserialize::<Value>()?;
53                if !value.is_array() {
54                    return Err(PropertyDataTypeError::ValueIsNotOfTheExpectedDataType(property.name.clone(), property.data_type, DataType::Array).into());
55                }
56                properties.insert(property.key().clone(), value);
57            }
58            DataType::Object => {
59                let value = field_arg_value.deserialize::<Value>()?;
60                if !value.is_object() {
61                    return Err(PropertyDataTypeError::ValueIsNotOfTheExpectedDataType(property.name.clone(), property.data_type, DataType::Object).into());
62                }
63                properties.insert(property.key().clone(), value);
64            }
65            DataType::Any => {
66                if let Ok(value) = field_arg_value.deserialize() {
67                    properties.insert(property.key().clone(), value);
68                }
69            }
70        }
71    }
72    Ok(properties)
73}

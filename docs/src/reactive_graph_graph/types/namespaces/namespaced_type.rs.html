40pub trait NamespacedTypeConstructor: FromStr<Err = NamespacedTypeParseError> {
41    /// Creates a new namespaced type.
42    fn new<NT: Into<NamespacedType>>(nt: NT) -> Self;
43
44    fn parse_namespace(namespace: &str) -> Result<Self, NamespacedTypeError>
45    where
46        Self: Sized,
47    {
48        Ok(Self::new(NamespacedType::from_str(namespace)?))
49    }
50
51    fn parse_optional_namespace(namespace: Option<String>) -> Result<Option<Self>, NamespacedTypeError>
52    where
53        Self: Sized,
54    {
55        match namespace {
56            None => Ok(None),
57            Some(namespace) => Self::parse_namespace(&namespace).map(Some),
58        }
59    }
60}

134impl NamespacedType {
135    /// Constructs a new namespaced type.
136    pub fn new<N: Into<Namespace>>(namespace: N) -> Result<NamespacedType, NamespacedTypeError> {
137        let namespace = namespace.into();
138        if !namespace.is_type() {
139            return Err(NamespacedTypeError::TypeNameMissing(namespace));
140        }
141        let Some(path) = namespace.parent() else {
142            return Err(NamespacedTypeError::MissingPathForType(namespace));
143        };
144        let Some(type_name) = namespace.last_segment() else {
145            return Err(NamespacedTypeError::TypeNameMissing(namespace));
146        };
147        Ok(NamespacedType { namespace, path, type_name })
148    }
149
150    /// Returns the fully qualified namespace of the namespaced type.
151    #[inline]
152    pub fn namespace(&self) -> Namespace {
153        NamespacedTypeGetter::namespace(self)
154    }
155
156    /// Returns the path of the namespaced type without the type name segment.
157    #[inline]
158    pub fn path(&self) -> Namespace {
159        NamespacedTypeGetter::path(self)
160    }
161
162    /// Returns the name of the namespaced type.
163    #[inline]
164    pub fn type_name(&self) -> NamespaceSegment {
165        NamespacedTypeGetter::type_name(self)
166    }
167
168    /// Returns the fully qualified type name in pascal case.
169    ///
170    /// For example, namespace1::namespace2::namespace3::TypeName will become
171    /// Namespace1Namespace2Namespace3TypeName
172    #[inline]
173    pub fn fully_qualified_type_name(&self) -> String {
174        NamespacedTypeGetter::fully_qualified_type_name(self)
175    }
176}

259    pub fn get_namespaces(&self) -> Namespaces {
260        let mut namespaces = Namespaces::new();
261        let namespaced_types = self.to_sorted_vec();
262        for namespaced_type in namespaced_types {
263            let path = namespaced_type.path();
264            namespaces.insert(path);
265        }
266        namespaces
267    }
268
269    /// Returns the paths and their parent paths of all types.
270    pub fn get_all_parent_paths_recursively(&self) -> Namespaces {
271        let mut namespaces = Namespaces::new();
272        for namespaced_type in self.0.iter() {
273            let mut path = namespaced_type.path();
274            loop {
275                namespaces.insert(path.clone());
276                path = match path.parent() {
277                    None => break,
278                    Some(path) => path,
279                };
280            }
281        }
282        namespaces
283    }

447        assert!(!NAMESPACED_TYPE_REGEX.is_match("namespace::"), "Last segment must be a type name and must not be empty");

451    fn namespaced_type_from_str_test() {
452        let path_segment_1 = r_namespace_path_segment();
453        let path_segment_2 = r_namespace_path_segment();
454        let type_name = r_namespace_type_name();
455        let ns = Namespace::try_new_top_level(&path_segment_1)
456            .and_then(|namespace| namespace.try_append(&path_segment_2))
457            .and_then(|namespace| namespace.try_append(&type_name))
458            .expect("Failed to construct namespace");
459        let nt = NamespacedType::new(&ns).expect("Failed to create Namespaced type");
460        assert_eq!(ns, nt.namespace());
461        assert_eq!(ns.parent().unwrap(), nt.path());
462        assert_eq!(ns.parent().unwrap().to_string(), nt.path().to_string());
463        assert_eq!(ns.last_segment().unwrap(), nt.type_name());
464        assert_eq!(ns.last_segment().unwrap().to_string(), nt.type_name().to_string());
465        assert_eq!(format!("{path_segment_1}::{path_segment_2}::{type_name}"), format!("{}", nt));
466        assert_eq!(ns.to_string(), nt.to_string());

5pub trait ComponentTypeIdContainer {
6    /// Returns true, if the type is composed with a component of the given type.
7    fn is_a(&self, ty: &ComponentTypeId) -> bool;
8
9    /// Returns true, if the type is composed with any of the given components.
10    fn is_any(&self, component_tys: &ComponentTypeIds) -> bool {
11        component_tys.iter().any(|ty| self.is_a(&ty))
12    }
13
14    /// Returns true, if the type is composed with every given components.
15    fn is_all(&self, component_tys: &ComponentTypeIds) -> bool {
16        component_tys.iter().all(|ty| self.is_a(&ty))
17    }
18
19    /// Adds the given component.
20    fn add_component<C: Into<ComponentTypeId>>(&self, ty: C) -> bool;
21
22    /// Adds the given components.
23    fn add_components<C: Into<ComponentTypeIds>>(&mut self, components_to_add: C);
24
25    /// Removes the component with the given type id.
26    fn remove_component(&self, ty: &ComponentTypeId) -> Option<ComponentTypeId>;
27
28    /// Removes the given components.
29    fn remove_components<C: Into<ComponentTypeIds>>(&mut self, components_to_remove: C);
30
31    fn get_components_cloned(&self) -> ComponentTypeIds;
32}

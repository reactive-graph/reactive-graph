6/// Reactive instance container for components.
7pub trait ComponentContainer {
8    /// Returns the component types of the container.
9    fn get_components(&self) -> ComponentTypeIds;
10
11    /// Adds a component to the container.
12    fn add_component(&self, ty: ComponentTypeId);
13
14    /// Adds a component to the container and initializes the reactive property instances.
15    fn add_component_with_properties(&self, component: &Component);
16
17    /// Removes a component from the container.
18    fn remove_component(&self, ty: &ComponentTypeId);
19
20    /// Returns true, if the reactive instance is composed with the given component.
21    fn is_a(&self, ty: &ComponentTypeId) -> bool;
22
23    /// Returns true, if the reactive instance is composed with all the given components.
24    fn is_all(&self, tys: &ComponentTypeIds) -> bool;
25}
26
27pub trait ComponentContainerGetter {

31    /// Returns the component type id.

33}

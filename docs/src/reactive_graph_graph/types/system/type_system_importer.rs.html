1use crate::Components;
2use crate::EntityTypes;
3use crate::FlowTypes;
4use crate::RelationTypes;
5use crate::TypeDefinitionImportError;
6use crate::TypeDefinitionImporter;
7use crate::TypeSystem;
8use std::path::PathBuf;
9
10impl TypeDefinitionImporter<TypeSystem> for TypeSystem {
11    fn import(path: PathBuf) -> Result<Self, TypeDefinitionImportError> {
12        if !path.exists() || !path.is_dir() {
13            return Err(TypeDefinitionImportError::ImportPathDoesNotExist(path));
14        }
15        Ok(TypeSystem::builder()
16            .components(Components::import(path.join("components")).unwrap_or_default())
17            .entity_types(EntityTypes::import(path.join("entities")).unwrap_or_default())
18            .relation_types(RelationTypes::import(path.join("relations")).unwrap_or_default())
19            .flow_types(FlowTypes::import(path.join("flows")).unwrap_or_default())
20            .build())
21    }
22}

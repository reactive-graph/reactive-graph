13/// A type which contains variables.
14pub trait VariablesContainer {
15    /// Returns true, if the type contains an variables with the given variable_name.
16    fn has_variable<S: Into<String>>(&self, variable_name: S) -> bool;
17
18    /// Returns true, if the type contains any of the given variables.
19    fn has_any_variables(&self, variable_names: &VariableNames) -> bool {
20        variable_names.iter().any(|variable_name| self.has_variable(variable_name))
21    }
22
23    /// Returns true, if the type contains all of the given variables.
24    fn has_all_variables(&self, variable_names: &VariableNames) -> bool {
25        variable_names.iter().all(|variable_name| self.has_variable(variable_name))
26    }
27
28    /// Returns the variable with the given variable_name.
29    fn get_variable<N: Into<String>>(&self, variable_name: N) -> Option<Variable>;
30
31    /// Adds the given variable.
32    fn add_variable<V: Into<Variable>>(&self, variable: V) -> Result<Variable, AddVariableError>;
33
34    /// Updates the variable with the given variable_name.
35    /// It's possible to rename the variable by using another name in the new variable than the provided variable_name.
36    fn update_variable<N: Into<String>, V: Into<Variable>>(&self, variable_name: N, variable: V) -> Result<Variable, UpdateVariableError>;
37
38    /// Removes the variable with the given variable_name.
39    fn remove_variable<S: Into<String>>(&self, variable_name: S) -> Result<Variable, RemoveVariableError>;
40
41    /// Merges the given variables with the existing variables.
42    fn merge_variables<V: Into<Variables>>(&mut self, variables_to_merge: V);
43
44    /// Merges only new variables into the own variables.
45    /// Returns true, if the variable already exists and the data type is divergent.
46    fn merge_non_existent_variables<V: Into<Variables>>(&self, variables_to_merge: V) -> DivergentPropertyTypes;
47
48    /// Returns all own properties.
49    fn get_own_variables_cloned(&self) -> Variables;
50}
51
52/// Collection of a type which contains variables.
53pub trait NamespacedTypeVariablesContainer<T, AddVariableError, UpdateVariableError, RemoveVariableError, MergeVariablesError> {
54    /// Adds a variable to the given type.
55    fn add_variable<V: Into<Variable>>(&self, ty: &T, variable: V) -> Result<Variable, AddVariableError>;
56
57    /// Updates the variable with the given name of the given type.
58    /// It's possible to rename the variable by using another name in the new variable than the provided variable_name.
59    fn update_variable<N: Into<String>, V: Into<Variable>>(&self, ty: &T, variable_name: N, variable: V) -> Result<Variable, UpdateVariableError>;
60
61    /// Remove the variable with the given name from the given type.
62    fn remove_variable<N: Into<String>>(&self, ty: &T, variable_name: N) -> Result<Variable, RemoveVariableError>;
63
64    /// Merges the given variables with the variables of the given type.
65    fn merge_variables<V: Into<Variables>>(&mut self, ty: &T, variables_to_merge: V) -> Result<(), MergeVariablesError>;
66}

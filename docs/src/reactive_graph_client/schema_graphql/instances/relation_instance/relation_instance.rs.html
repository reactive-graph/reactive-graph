25    relation_type: RelationType,

34impl RelationInstance {
35    pub fn ty(&self) -> Result<RelationTypeId, RelationInstanceTypeIdError> {
36        RelationTypeId::from_str(&self.relation_type._type).map_err(RelationInstanceTypeIdError::NamespacedTypeParseError)
37    }
38
39    pub fn instance_ty(&self) -> Result<RelationInstanceTypeId, RelationInstanceTypeIdError> {
40        Ok(RelationInstanceTypeId::new(
41            self.ty()?,
42            InstanceId::parse_named(&self.instance_id).map_err(RelationInstanceTypeIdError::InstanceIdError)?,
43        ))
44    }
45}

47impl TryFrom<RelationInstance> for reactive_graph_graph::RelationInstance {
48    type Error = InvalidRelationInstanceError;
49
50    fn try_from(relation_instance: RelationInstance) -> Result<Self, Self::Error> {
51        let ty = relation_instance
52            .instance_ty()
53            .map_err(InvalidRelationInstanceError::InvalidRelationInstanceTypeId)?;
54        Ok(reactive_graph_graph::RelationInstance {
55            outbound_id: relation_instance.outbound.id.into(),
56            ty,
57            inbound_id: relation_instance.inbound.id.into(),
58            name: relation_instance.name.clone(),
59            description: relation_instance.description.clone(),
60            properties: PropertyInstances(relation_instance.properties).into(),
61            components: reactive_graph_graph::Components::try_from(Components(relation_instance.components))
62                .map_err(InvalidRelationInstanceError::InvalidComponent)?
63                .type_ids(),
64            // TODO: implement!
65            extensions: Default::default(),
66        })
67    }
68}

80impl TryFrom<RelationInstances> for reactive_graph_graph::RelationInstances {
81    type Error = InvalidRelationInstanceError;
82
83    fn try_from(relations: RelationInstances) -> Result<Self, Self::Error> {
84        let relation_instances = reactive_graph_graph::RelationInstances::new();
85        for relation_instance in relations.0 {
86            relation_instances.push(reactive_graph_graph::RelationInstance::try_from(relation_instance)?);
87        }
88        Ok(relation_instances)
89    }

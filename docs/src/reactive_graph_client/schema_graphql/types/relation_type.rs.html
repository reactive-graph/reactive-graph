76impl TryFrom<RelationType> for reactive_graph_graph::RelationType {
77    type Error = InvalidRelationTypeError;
78
79    fn try_from(relation_type: RelationType) -> Result<Self, Self::Error> {
80        let components = Components(relation_type.components.clone());
81        Ok(reactive_graph_graph::RelationType {
82            outbound_type: relation_type.get_outbound_type()?,
83            ty: RelationTypeId::from_str(&relation_type._type).map_err(InvalidRelationTypeError::InvalidRelationType)?,
84            inbound_type: relation_type.get_inbound_type()?,
85            description: relation_type.description,
86            components: reactive_graph_graph::Components::try_from(components)
87                .map_err(InvalidRelationTypeError::InvalidComponent)?
88                .type_ids(),
89            properties: reactive_graph_graph::PropertyTypes::try_from(PropertyTypes(relation_type.properties))
90                .map_err(InvalidRelationTypeError::InvalidPropertyType)?,
91            extensions: reactive_graph_graph::Extensions::try_from(Extensions(relation_type.extensions)).map_err(InvalidRelationTypeError::InvalidExtension)?,
92        })
93    }
94}

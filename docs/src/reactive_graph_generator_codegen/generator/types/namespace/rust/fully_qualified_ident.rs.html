15pub trait NamespacedTypeIdent {
16    fn ident_of_type<IdentType: AssocTypeDefinitionIdentType + ?Sized>(&self) -> Result<Ident, CodeGenerationError>;
17}
18
19impl<TY: NamespacedTypeGetter + TypeDefinitionGetter> NamespacedTypeIdent for TY {
20    fn ident_of_type<IdentType: AssocTypeDefinitionIdentType + ?Sized>(&self) -> Result<Ident, CodeGenerationError> {
21        Ok(TypeDefinitionIdent::<IdentType>::new(self).deref().clone())
22    }
23}
24
25pub trait FullyQualifiedNamespacedTypeIdent: NamespacedTypeIdent {
26    fn crate_name(&self, resolver: &TypeResolver) -> Result<Ident, CodeGenerationError>;
27    fn fully_qualified_module(&self) -> Result<TokenStream, CodeGenerationError>;
28    fn fully_qualified_ident_of_type<IdentType: AssocTypeDefinitionIdentType + ?Sized>(
29        &self,
30        resolver: &TypeResolver,
31    ) -> Result<TokenStream, CodeGenerationError>;
32}
33
34impl<TY: NamespacedTypeGetter + TypeDefinitionGetter> FullyQualifiedNamespacedTypeIdent for TY {
35    fn crate_name(&self, resolver: &TypeResolver) -> Result<Ident, CodeGenerationError> {
36        let crate_name = resolver.crate_name(self).ok_or(CodeGenerationError::NamespaceParseError(self.namespace()))?;
37        if crate_name.is_empty() || crate_name == "build_script_build" {
38            Ok(Ident::new("crate", Span::call_site()))
39        } else {
40            Ok(Ident::new(&crate_name.to_case(Case::Snake), Span::call_site()))
41        }
42    }
43
44    fn fully_qualified_module(&self) -> Result<TokenStream, CodeGenerationError> {
45        let path = self.path();
46        let fully_qualified_module = format!("{}::{}", path.to_string(), self.type_name().to_case(Case::Snake));
47        str::parse::<TokenStream>(&fully_qualified_module).map_err(|_| CodeGenerationError::NamespaceParseError(path))
48    }
49
50    fn fully_qualified_ident_of_type<IdentType: AssocTypeDefinitionIdentType + ?Sized>(
51        &self,
52        resolver: &TypeResolver,
53    ) -> Result<TokenStream, CodeGenerationError> {
54        let crate_ident = self.crate_name(resolver)?;
55        let fully_qualified_module = self.fully_qualified_module()?;
56        let ident = self.ident_of_type::<IdentType>()?;
57        Ok(quote! {
58            #crate_ident::#fully_qualified_module::#ident
59        })
60    }
61}

12pub fn generate_const_type_id<TY: TypeDefinitionGetter>(type_: &TY) -> TokenStream {
13    let type_definition = type_.type_definition();
14    let const_namespace_ident = ConstNamespaceIdent::new(type_);
15    let type_id_type_ident = TypeTypeIdIdent::new(type_);
16    let const_type_id_ident = ConstTypeIdIdent::new(type_);
17    let doc_comment = Rust::multiline_doc_comment(const_type_id_doc_comment(&type_definition));
18
19    quote! {
20        #[doc(newline)]
21        #doc_comment
22        pub static #const_type_id_ident: std::sync::LazyLock<reactive_graph_graph::#type_id_type_ident> = std::sync::LazyLock::new(
23            || std::str::FromStr::from_str(#const_namespace_ident).unwrap()
24        );
25    }
26}
27
28#[inline]
29fn const_type_id_doc_comment(type_definition: &TypeDefinition) -> String {
30    format!(
31        "The [type identifier]() of {} `{}`.\n\n### Namespace\n\nThe fully qualified namespace is\n `{}`",
32        type_definition.type_id_type.full_name(),
33        type_definition.type_name().to_string(),
34        type_definition.namespace().to_string(),
35    )
36}

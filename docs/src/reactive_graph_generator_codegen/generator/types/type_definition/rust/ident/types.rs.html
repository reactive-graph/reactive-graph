9pub enum TypeDefinitionIdentTypes {

23impl TypeDefinitionIdentTypes {
24    pub fn into_ident<TY: TypeDefinitionGetter, IdentType: AssocTypeDefinitionIdentType + ?Sized>(self, ty: &TY) -> TypeDefinitionIdent<IdentType> {
25        TypeDefinitionIdent::<IdentType>::new(ty)
26    }
27
28    pub fn casing(&self) -> Case<'_> {
29        match self {
30            TypeDefinitionIdentTypes::TypeTypeId => Case::UpperCamel,
31            TypeDefinitionIdentTypes::TypeType => Case::UpperCamel,
32            TypeDefinitionIdentTypes::ConstNamespace => Case::UpperSnake,
33            TypeDefinitionIdentTypes::ConstTypeId => Case::UpperSnake,
34            TypeDefinitionIdentTypes::TypeProperties => Case::UpperCamel,
35            TypeDefinitionIdentTypes::TypePropertiesIterator => Case::UpperCamel,
36            TypeDefinitionIdentTypes::ConstComponents => Case::UpperSnake,
37            TypeDefinitionIdentTypes::ConstExtensions => Case::UpperSnake,
38            TypeDefinitionIdentTypes::ConstType => Case::UpperSnake,
39            TypeDefinitionIdentTypes::Type => Case::UpperCamel,
40            TypeDefinitionIdentTypes::TypeValue => Case::Snake,
41        }
42    }
43
44    pub fn postfix(&self) -> String {
45        match self {
46            TypeDefinitionIdentTypes::TypeTypeId => "TypeId".to_string(),
47            TypeDefinitionIdentTypes::TypeType => String::new(), // TODO: "Type".to_string(),
48            TypeDefinitionIdentTypes::ConstNamespace => "Namespace".to_string(),
49            TypeDefinitionIdentTypes::ConstTypeId => String::new(),
50            TypeDefinitionIdentTypes::TypeProperties => "Properties".to_string(),
51            TypeDefinitionIdentTypes::TypePropertiesIterator => "PropertiesIterator".to_string(),
52            TypeDefinitionIdentTypes::ConstComponents => "Components".to_string(),
53            TypeDefinitionIdentTypes::ConstExtensions => "Extensions".to_string(),
54            TypeDefinitionIdentTypes::ConstType => "Type".to_string(),
55            TypeDefinitionIdentTypes::Type => String::new(),
56            TypeDefinitionIdentTypes::TypeValue => String::new(),
57        }
58    }
59
60    pub fn name(&self, type_definition: &TypeDefinition) -> String {
61        match self {
62            TypeDefinitionIdentTypes::TypeTypeId => match type_definition.type_id_type {
63                TypeIdType::Behaviour => "Behaviour",
64                TypeIdType::Component => "Component",
65                TypeIdType::EntityType => "Entity",
66                TypeIdType::Extension => "Extension",
67                TypeIdType::RelationType => "Relation",
68                TypeIdType::FlowType => "Flow",
69            }
70            .to_string(),
71            TypeDefinitionIdentTypes::TypeType => match type_definition.type_id_type {
72                TypeIdType::Behaviour => "BehaviourType",
73                TypeIdType::Component => "Component", // fix this to make postfixes safe to use!
74                TypeIdType::EntityType => "EntityType",
75                TypeIdType::Extension => "ExtensionType",
76                TypeIdType::RelationType => "RelationType",
77                TypeIdType::FlowType => "FlowType",
78            }
79            .to_string(),
80            TypeDefinitionIdentTypes::ConstNamespace
81            | TypeDefinitionIdentTypes::ConstTypeId
82            | TypeDefinitionIdentTypes::TypeProperties
83            | TypeDefinitionIdentTypes::TypePropertiesIterator
84            | TypeDefinitionIdentTypes::ConstComponents
85            | TypeDefinitionIdentTypes::ConstExtensions
86            | TypeDefinitionIdentTypes::ConstType
87            | TypeDefinitionIdentTypes::Type
88            | TypeDefinitionIdentTypes::TypeValue => type_definition.type_name().to_string(),
89        }
90    }

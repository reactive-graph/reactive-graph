18pub fn generate_impl_component_traits<TY: TypeDefinitionGetter + ComponentTypeIdContainer + TypeDescriptionGetter + GenerateDocumentation<TY>>(
19    type_: &TY,
20    resolver: &TypeResolver,
21) -> Result<TokenStream, CodeGenerationError> {
22    let type_name_ident = TypeIdent::new(type_);
23    let component_tys = type_.get_components_cloned();
24    let mut components = resolver.components(&component_tys)?.to_vec();
25    components.sort();
26    let mut impl_component_traits = Vec::new();
27    for component in components {
28        let fully_qualified_component_ident = component.ty.fully_qualified_ident_of_type::<TypeIdent>(resolver)?;
29        let mut properties = component.properties.to_vec();
30        properties.sort();
31        let mut properties_token_stream = Vec::new();
32        for property in properties {
33            properties_token_stream.push(generate_property_instance_getter_impl(&property, resolver, Visibility::Private));
34            if property.mutability == Mutability::Mutable {
35                properties_token_stream.push(generate_property_instance_setter_impl(&property, resolver, Visibility::Private));
36            }
37        }
38        let impl_component_trait = quote! {
39            impl #fully_qualified_component_ident for #type_name_ident {
40                #(#properties_token_stream)*
41            }
42        };
43        impl_component_traits.push(impl_component_trait);
44    }
45    Ok(quote! {
46        #(#impl_component_traits)*
47    })
48}

20pub fn generate_impl_trait_property_instance_getter_and_setter_method_signatures<
21    TY: TypeDefinitionGetter + PropertyTypeContainer + TypeDescriptionGetter + GenerateDocumentation<TY>,
22>(
23    type_: &TY,
24    resolver: &TypeResolver,
25) -> TokenStream {
26    let type_name_ident = TypeIdent::new(type_);
27    let mut token_stream = Vec::new();
28    let mut properties = type_.get_own_properties_cloned().to_vec();
29    properties.sort();
30    for property in properties.into_iter() {
31        token_stream.push(generate_property_instance_getter_method_signatures(&property, resolver));
32        if property.mutability == Mutability::Mutable {
33            token_stream.push(generate_property_instance_setter_method_signatures(&property, resolver));
34        }
35    }
36    let doc_comment = Rust::multiline_doc_comment(trait_doc_comment(type_, resolver));
37    quote! {
38        #[doc(newline)]
39        #doc_comment
40        pub trait #type_name_ident {
41            #(#token_stream)*
42        }
43    }
44}

75        .generate_documentation(&DocumentationConfig::new_from_preset(DocumentationConfigPreset::Short), resolver)

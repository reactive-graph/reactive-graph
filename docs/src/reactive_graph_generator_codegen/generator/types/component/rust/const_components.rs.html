18pub fn generate_const_components<TY: TypeDefinitionGetter + ComponentTypeIdContainer>(
19    type_: &TY,
20    resolver: &TypeResolver,
21) -> Result<TokenStream, CodeGenerationError> {
22    let const_components_ident = ConstComponentsIdent::new(type_);
23
24    let component_tys = type_.get_components_cloned();
25    let doc_comment = Rust::multiline_doc_comment(const_components_doc_comment(&component_tys, resolver));
26
27    let mut components = component_tys.to_vec();
28    components.sort();
29    let mut component_idents = Vec::new();
30    for ty in components.iter() {
31        component_idents.push(ty.fully_qualified_ident_of_type::<ConstTypeIdIdent>(resolver)?);
32    }
33
34    Ok(quote! {
35        #[doc(newline)]
36        #doc_comment
37        pub static #const_components_ident: std::sync::LazyLock<reactive_graph_graph::ComponentTypeIds> = std::sync::LazyLock::new(|| {
38            reactive_graph_graph::ComponentTypeIds::new()
39                #(.component(std::ops::Deref::deref(&#component_idents)))*
40        });
41    })
42}

44#[inline]
45fn const_components_doc_comment<TY: ComponentTypeIdContainer + GenerateDocumentation<TY>>(type_: &TY, resolver: &TypeResolver) -> String {
46    type_
47        .generate_documentation(&DocumentationConfig::new_from_preset(DocumentationConfigPreset::Short), resolver)
48        .map(|documentation| format!(" {}", documentation.to_string()))
49        .unwrap_or_default()
50}

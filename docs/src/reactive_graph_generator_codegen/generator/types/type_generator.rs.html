12pub trait TypeGeneratorForTarget: TypeGenerator<Java> + TypeGenerator<Rust> {
13    fn generate_type_for_target(
14        &self,
15        target: CodeGenerationTargets,
16        config: &CodeGenerationConfig,
17        resolver: &TypeResolver,
18    ) -> Result<CodeGenerationResult, CodeGenerationError> {
19        match target {
20            CodeGenerationTargets::Java => TypeGenerator::<Java>::generate_type(self, config, resolver),
21            CodeGenerationTargets::Rust => TypeGenerator::<Rust>::generate_type(self, config, resolver),
22            _ => Err(CodeGenerationError::TargetNotSupported(target)),
23        }
24    }
25}

27impl<TY> TypeGeneratorForTarget for TY where TY: TypeGenerator<Java> + TypeGenerator<Rust> {}

29pub trait TypeGenerator<Target: CodeGenerationTarget> {
30    fn generate_type(&self, config: &CodeGenerationConfig, resolver: &TypeResolver) -> Result<CodeGenerationResult, CodeGenerationError>;
31
32    fn register_child(
33        &self,
34        config: &CodeGenerationConfig,
35        _resolver: &TypeResolver,
36        _child_namespace: &Namespace,
37    ) -> Result<CodeGenerationResult, CodeGenerationError> {
38        let path = self.absolute_target_path(config)?;
39        Ok(CodeGenerationResult { path })
40    }
41
42    fn register_root(&self, config: &CodeGenerationConfig, _resolver: &TypeResolver) -> Result<CodeGenerationResult, CodeGenerationError> {
43        Ok(CodeGenerationResult { path: config.base_path() })
44    }
45
46    fn absolute_target_path(&self, config: &CodeGenerationConfig) -> Result<PathBuf, CodeGenerationError> {
47        Ok(config.as_ref().join(self.relative_target_path()?))
48    }
49    fn relative_target_path(&self) -> Result<PathBuf, CodeGenerationError>;
50}

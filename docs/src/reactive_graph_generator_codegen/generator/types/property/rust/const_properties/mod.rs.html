21pub fn generate_const_properties<TY: TypeDefinitionGetter + PropertyTypeContainer>(
22    type_: &TY,
23    resolver: &TypeResolver,
24    properties: &Vec<PropertyType>,
25) -> Result<TokenStream, CodeGenerationError> {
26    let type_properties_ident = TypePropertiesIdent::new(type_);
27    let const_properties_enum = const_enum::generate_const_properties_enum(type_, resolver, &type_properties_ident, &properties)?;
28    let as_ref_const_properties = as_ref::generate_as_ref_const_properties(&type_properties_ident, &properties);
29    let convert_const_properties_into_static_str = static_str::generate_convert_const_properties_into_static_str(&type_properties_ident, &properties);
30    let convert_const_properties_into_string = into_string::generate_convert_const_properties_into_string(&type_properties_ident, &properties);
31    let convert_const_properties_into_property_type = property_type::generate_convert_const_properties_into_property_type(&type_properties_ident, &properties);
32    let into_iter = into_iter::generate_into_iter(type_, &type_properties_ident, &properties);
33    let const_properties_display = display::generate_const_properties_display(&type_properties_ident, &properties);
34
35    Ok(quote! {
36        #const_properties_enum
37        #as_ref_const_properties
38        #convert_const_properties_into_static_str
39        #convert_const_properties_into_string
40        #convert_const_properties_into_property_type
41        #into_iter
42        #const_properties_display
43        // TODO: Generate property extensions
44        // impl { pub fn extensions() -> Extensions { ... } }
45    })
46}
